  • Get player`s kills
public function getData(Player $player): \mystats\utils\Data {
    $api = \mystats\MyStats::getAPI();
    return $api->getPlayerData($player);
}


public function getKills(Player $player): int {
    $data = $this->getData($player);
    return $data->getKills();
}
